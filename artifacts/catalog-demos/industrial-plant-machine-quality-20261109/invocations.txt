npx tsx scripts/buildAssetCatalog.ts
python3 artifacts/catalog-demos/industrial-plant-machine-quality-20261109/verify_catalog_index.py

python3 scripts/checkCatalogQuality.py --kit industrial-plant
python3 scripts/checkCatalogQuality.py

npx tsx scripts/buildCatalogPages.ts /tmp/kernelcad-industrial-plant-machine-pages-20261109
python3 artifacts/catalog-demos/industrial-plant-machine-quality-20261109/verify_local_pages.py /tmp/kernelcad-industrial-plant-machine-pages-20261109
