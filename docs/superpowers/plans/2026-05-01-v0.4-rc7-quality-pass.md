```typescript
// src/intent/kernelError.ts
//
// `KernelError` is the kernelCAD-thrown exception type. Carrying a structured
// `code` field lets the script-runtime exception path emit a `CompilerDiagnostic`
// with the same code that `whyDidThisFail` registers a hint for, instead of
// the generic `cli.script.exception` fallback.
//
// All kernel-side throws that need to be agent-discoverable (path label
// validation, selectEdge ambiguity, future kernel-internal validation) should
// throw `KernelError` rather than plain `Error`.

export class KernelError extends Error {
  readonly code: string;

  constructor(code: string, message: string) {
    super(message);
    this.code = code;
    this.name = 'KernelError';
  }
}

export function isKernelError(e: unknown): e is KernelError {
  return e instanceof KernelError || (
    typeof e === 'object' && e !== null && 'code' in e && (e as { name?: string }).name === 'KernelError'
  );
}
```

The structural `isKernelError` fallback handles the case where a `KernelError` thrown inside the `vm` sandbox in `runIsolated` reaches the host with a different `Error` prototype (cross-realm). The tests in Step 7 verify this works in practice.

```typescript
// src/script-runtime/kernelErrorToDiagnostic.ts
//
// Converts a script-runtime exception into a `CompilerDiagnostic`. KernelError
// carries its own diagnostic code; everything else gets `cli.script.exception`.
import type { CompilerDiagnostic } from '../diagnostics/diagnostic';
import { isKernelError } from '../intent/kernelError';

export function kernelErrorToDiagnostic(e: unknown): CompilerDiagnostic {
  if (isKernelError(e)) {
    return {
      target: 'export-occt',
      code: (e as { code: string }).code,
      severity: 'error',
      message: e instanceof Error ? e.message : String(e),
    };
  }
  const msg = e instanceof Error ? e.message : String(e);
  return {
    target: 'export-occt',
    code: 'cli.script.exception',
    severity: 'error',
    message: msg,
  };
}
```

```typescript
function resolveLabeledFace(
  record: FeatureRecord,
  base: OcctBackend,
  label: string,
  records: readonly FeatureRecord[] | undefined,
): { face: import('replicad').Face } | { error: CompilerDiagnostic } {
  // Reuse labelToEdgeQuery to compute the probe point, then find the matching
  // face on the lowered shape. Same probe geometry as the edge path: midpoint
  // of the labeled segment, lifted to z = depth/2.
  const probe = labelToEdgeQuery(record, base, label, records);
  if ('error' in probe) return probe;

  // probe.query has the `within` bbox + tolerance — resolve face by checking
  // which face's centroid sits in or near that bbox.
  const allFaces = (base.getReplicadShape() as unknown as { faces: import('replicad').Face[] }).faces;
  const w = probe.query.within;
  if (!w) {
    return {
      error: {
        target: 'export-occt',
        code: 'feature.face-feature.label-not-resolvable',
        featureId: record.id,
        severity: 'error',
        message: `Label '${label}': cannot derive face probe (no within bbox).`,
      },
    };
  }
  const matched = allFaces.filter(f => {
    const c = f.center;
    return (w.xMin === undefined || c.x >= w.xMin) &&
           (w.xMax === undefined || c.x <= w.xMax) &&
           (w.yMin === undefined || c.y >= w.yMin) &&
           (w.yMax === undefined || c.y <= w.yMax) &&
           (w.zMin === undefined || c.z >= w.zMin) &&
           (w.zMax === undefined || c.z <= w.zMax);
  });
  if (matched.length === 0) {
    return {
      error: {
        target: 'export-occt',
        code: 'feature.face-feature.label-not-resolvable',
        featureId: record.id,
        severity: 'error',
        message: `Label '${label}' resolved to a probe bbox that contained no face centroid.`,
      },
    };
  }
```

```typescript
  /**
   * Lower this Shape eagerly — runs recompute against the records up to and
   * including this Shape, returns the resulting OcctBackend so script-runtime
   * helpers like `selectEdges` can introspect the lowered geometry.
   *
   * Most agents won't call this directly. It's invoked implicitly when an
   * agent calls `selectEdges(myShape, ...)` from a `.kcad.ts` script.
   */
  async lower(): Promise<import('../backends/occt/occtBackend').OcctBackend> {
    const { RecomputeEngine } = await import('../compute/recomputeEngine');
    const { OcctLowerer } = await import('../backends/occt/occtLowerer');
    const { OcctBackend, initOcct } = await import('../backends/occt/occtBackend');
    await initOcct();
    const engine = new RecomputeEngine(new OcctLowerer());
    const records = this.session.getRecords();
    const r = await engine.run(records as readonly import('../intent/featureRecord').FeatureRecord[]);
    const shape = r.shapes.get(this.id);
    if (!shape) {
      throw new Error(`Shape.lower(): shape '${this.id}' not lowered (check upstream diagnostics).`);
    }
    if (!(shape instanceof OcctBackend)) {
      throw new Error(`Shape.lower(): shape '${this.id}' is not an OcctBackend.`);
    }
    return shape;
  }
```
