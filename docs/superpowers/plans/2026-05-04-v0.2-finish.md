- [ ] **Step 6: Implement resolution in `resolveFaceQuery`.**

In `src/backends/occt/edgeQueries.ts`, find `resolveFaceQuery` (around line 266). Add the four filters AFTER the existing AND-combine pipeline. Each filter is a `.filter(...)` step on the running `Face[]` accumulator. Pseudocode for the additions:

```typescript
// existing filters: parallelTo, inPlane, ofSurfaceType, containsPoint, atZ/X/Y, near, tolerance ...

if (query.byNormal !== undefined) {
  const axisMap: Record<string, [number, number, number]> = {
    'X': [1, 0, 0], '-X': [-1, 0, 0],
    'Y': [0, 1, 0], '-Y': [0, -1, 0],
    'Z': [0, 0, 1], '-Z': [0, 0, -1],
  };
  const target = axisMap[query.byNormal];
  if (!target) {
    throw new Error(`FaceQuery.byNormal: invalid axis '${query.byNormal}'`);
  }
  result = result.filter(f => {
    const n = f.normal; // assume Face exposes .normal as { x, y, z }
    const dot = n.x * target[0] + n.y * target[1] + n.z * target[2];
    return dot > 1 - (query.tolerance ?? 0.01);
  });
}

if (query.minArea !== undefined) {
  result = result.filter(f => faceArea(f) >= query.minArea!);
}
if (query.maxArea !== undefined) {
  result = result.filter(f => faceArea(f) <= query.maxArea!);
}

if (query.boundingBoxIn !== undefined) {
  const bbox = query.boundingBoxIn;
  result = result.filter(f => {
    const bb = faceBBox(f); // { xMin, xMax, yMin, yMax, zMin, zMax }
    return (
      (bbox.xMin === undefined || bb.xMin >= bbox.xMin) &&
      (bbox.xMax === undefined || bb.xMax <= bbox.xMax) &&
      (bbox.yMin === undefined || bb.yMin >= bbox.yMin) &&
      (bbox.yMax === undefined || bb.yMax <= bbox.yMax) &&
      (bbox.zMin === undefined || bb.zMin >= bbox.zMin) &&
      (bbox.zMax === undefined || bb.zMax <= bbox.zMax)
    );
  });
}
```

If `faceArea` and `faceBBox` helpers don't exist, add them as private helpers in the same file. They wrap Replicad's existing face mesh / bounding-box accessors. If unsure of the exact accessor name on Replicad's `Face`, search `node_modules/replicad/dist/replicad.d.ts` for `Face` interface methods (`area`, `getBoundingBox`, etc.) and use them directly.

- [ ] **Step 5: Add the validation helper.**

Add a private function in `src/capture/captureSession.ts`:

```typescript
import { KernelError } from '../intent/kernelError';

function validateFaceLabels(
  raw: unknown,
  shapeKind: string,
  featureId: string,
): FaceLabelsMap | undefined {
  if (raw === undefined) return undefined;
  if (typeof raw !== 'object' || raw === null || Array.isArray(raw)) {
    throw new KernelError({
      code: 'capture.faceLabels.invalid-shape',
      featureId,
      message: `faceLabels must be an object map, got ${typeof raw}`,
    });
  }
  const result: FaceLabelsMap = {};
  for (const [label, value] of Object.entries(raw)) {
    if (typeof label !== 'string' || label.length === 0) {
      throw new KernelError({
        code: 'capture.faceLabels.invalid-key',
        featureId,
        message: `faceLabels key must be a non-empty string`,
      });
    }
    if (typeof value === 'string') {
      // canonical alias — validate against the canonical set
      if (!['top','bottom','left','right','front','back'].includes(value)) {
        throw new KernelError({
          code: 'capture.faceLabels.invalid-value',
          featureId,
          message: `faceLabels['${label}'] = '${value}' is not a canonical face name`,
        });
      }
      result[label] = value as never;
    } else if (typeof value === 'object' && value !== null) {
      // query — accept as FaceQuery (full structural validation deferred to resolution time)
      result[label] = value as FaceQuery;
    } else {
      throw new KernelError({
        code: 'capture.faceLabels.invalid-value',
        featureId,
        message: `faceLabels['${label}'] must be a canonical face string or a FaceQuery object, got ${typeof value}`,
      });
    }
  }
  return result;
}
```

import type { FaceLabelsMap } from '../../intent/featureRecord';

interface MetadataLabelHit {
  /** The originating feature whose metadata.faceLabels declared this label. */
  origin: FeatureRecord;
  /** The resolved value: a canonical face name or a FaceQuery descriptor. */
  resolved: CanonicalFace | FaceQuery;
}

function findFaceLabelInMetadata(
  records: readonly FeatureRecord[],
  consumer: FeatureRecord,
  label: string,
): { hit: MetadataLabelHit } | { error: CompilerDiagnostic } {
  // Walk records backward from the consumer; collect every upstream feature
  // whose metadata.faceLabels[label] is defined.
  const hits: MetadataLabelHit[] = [];
  for (const rec of records) {
    if (rec.id === consumer.id) break; // only upstream
    const fl = (rec.metadata as { faceLabels?: FaceLabelsMap } | undefined)?.faceLabels;
    if (fl && Object.prototype.hasOwnProperty.call(fl, label)) {
      hits.push({ origin: rec, resolved: fl[label] });
    }
  }
  if (hits.length === 0) {
    return {
      error: {
        target: 'export-occt',
        code: 'feature.label.unknown-name',
        featureId: consumer.id,
        severity: 'error',
        message: `Label '${label}' was not declared by any upstream feature's faceLabels.`,
      },
    };
  }
  if (hits.length > 1) {
    return {
      error: {
        target: 'export-occt',
        code: 'feature.label.collision',
        featureId: consumer.id,
        severity: 'error',
        message: `Label '${label}' is declared by multiple upstream features: ${hits.map(h => h.origin.id).join(', ')}. Each label must be unique within the scope visible to a consumer.`,
      },
    };
  }
  return { hit: hits[0] };
}
